SAP_BAPI_OrderCallout.cls — Production Apex REST / JSON 
public with sharing class SAP_BAPI_OrderCallout {

  // Invoked by Record-Triggered Flow on Order Activation
  @InvocableMethod(label='Sync Order to SAP SD')
  public static void dispatchOrderToSAP(List<Id> orderIds) {
    System.enqueueJob(new SAP_OrderQueueable(orderIds));
  }

  public static HttpResponse callSapRfcEndpoint(Order ord) {
    HttpRequest req = new HttpRequest();
    req.setEndpoint('callout:SAP_RFC_Gateway/sap/opu/odata/SAP/ZSD_ORDER_SRV');
    req.setMethod('POST');
    req.setHeader('Content-Type', 'application/json');
    req.setHeader('x-csrf-token', 'Fetch');

    // Build SAP BAPI_SALESORDER_CREATEFROMDAT2 Payload
    Map<String, Object> sapPayload = new Map<String, Object>{
      'DOC_TYPE'   => 'ZOR',
      'SALES_ORG'  => '1000',
      'KUNNR'      => ord.Account.SAP_Customer_Code__c,
      'NET_VALUE'  => ord.TotalAmount,
      'SF_ORD_ID'  => ord.Id
    };

    req.setBody(JSON.serialize(sapPayload));
    Http http = new Http();
    HttpResponse res = http.send(req);

    if (res.getStatusCode() == 201) {
      SAP_ResponseWrapper wrap = (SAP_ResponseWrapper) 
        JSON.deserialize(res.getBody(), SAP_ResponseWrapper.class);
      ord.SAP_Sales_Order_No__c = wrap.VBELN;
      ord.Sync_Status__c = 'Synced with SAP';
      update ord;
    }
    return res;
  }
}
